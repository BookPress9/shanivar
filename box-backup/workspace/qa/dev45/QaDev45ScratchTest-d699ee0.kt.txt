package com.shanivar.nativeapp

import okhttp3.mockwebserver.*
import org.junit.Assert.*
import org.junit.Test

class QaDev45ScratchTest {
    private fun srv(code: Int, match: String) = MockWebServer().apply {
        dispatcher = object : Dispatcher() {
            override fun dispatch(r: RecordedRequest) =
                if (r.path!!.contains(match)) MockResponse().setResponseCode(code).setBody(if (code == 404) """{"code":"PGRST205"}""" else "")
                else MockResponse().setResponseCode(201)
        }
    }
    private fun hol(code: Int) = srv(code, "/orbit_events").use { server ->
        val rig = SyncRig(server)
        rig.outbox.enqueue("orbit", "o1", """{"id":"o1"}""")
        rig.outbox.enqueue("attempt", "a1", """{"id":"a1"}""")
        rig.outbox.enqueue("self_report", "s1", """{"id":"s1"}""")
        val o1 = rig.flusher.flush()
        println("QA-HOL$code flush1=$o1 pending=" + rig.outbox.pending.map { it.key })
        assertEquals("items behind the head go out on the first flush", listOf("orbit:o1"), rig.outbox.pending.map { it.key })
        repeat(3) { rig.flusher.flush() }
        assertTrue("still retrying after 4", rig.outbox.deadLetters.isEmpty())
        val last = rig.flusher.flush()
        println("QA-HOL$code flush5=$last dead=" + rig.outbox.deadLetters + " requests=" + server.requestCount)
        assertTrue(rig.outbox.pending.isEmpty())
        assertEquals("retry_limit", rig.outbox.deadLetters.single().code)
        assertEquals(code, rig.outbox.deadLetters.single().status)
        assertEquals(7, server.requestCount) // 5 x o1 + a1 + s1
    }
    @Test fun persistent404HeadOfLine() = hol(404)
    @Test fun persistent408HeadOfLine() = hol(408)
    @Test fun routine404ParksNeverDeadLetters() = srv(404, "/rpc/put_routine").use { server ->
        val rig = SyncRig(server)
        rig.outbox.enqueue("routine", "current", """{"id":"current","schema_version":3,"plan":{}}""")
        rig.outbox.enqueue("attempt", "a1", """{"id":"a1"}""")
        repeat(6) { rig.flusher.flush() }
        println("QA-ROUTINE pending=" + rig.outbox.pending.map { it.key to it.parked } + " dead=" + rig.outbox.deadLetters)
        assertTrue(rig.outbox.deadLetters.isEmpty())
        assertNotNull(rig.outbox.pending.single { it.key == "routine:current" }.parked)
        assertTrue(rig.outbox.pending.none { it.key == "attempt:a1" })
    }
    @Test fun serverOutage500StillStopsWholeQueue() = srv(500, "/orbit_events").use { server ->
        val rig = SyncRig(server)
        rig.outbox.enqueue("orbit", "o1", """{"id":"o1"}"""); rig.outbox.enqueue("attempt", "a1", """{"id":"a1"}""")
        repeat(7) { rig.flusher.flush() }
        assertEquals(2, rig.outbox.pending.size); assertTrue(rig.outbox.deadLetters.isEmpty())
    }
}
